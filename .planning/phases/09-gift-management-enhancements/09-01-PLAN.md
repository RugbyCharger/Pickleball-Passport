**cancelGift:**
```typescript
cancelGift: protectedProcedure
  .input(z.object({ bookingId: z.string() }))
  .mutation(async ({ ctx, input }) => {
    // 1. Find booking and verify purchaser owns it
    const booking = await ctx.db.booking.findUnique({
      where: { id: input.bookingId },
    })
    if (!booking?.isGift || booking.giftPurchaserId !== ctx.user.id) {
      throw new TRPCError({ code: 'FORBIDDEN', message: 'Not authorized' })
    }
    // 2. Verify status is PENDING (only PENDING can be cancelled)
    if (booking.giftStatus !== 'PENDING') {
      throw new TRPCError({
        code: 'BAD_REQUEST',
        message: 'Only pending gifts can be cancelled'
      })
    }
    // 3. Execute transition to CANCELLED
    const result = await transitionGiftState(
      booking.id,
      GiftState.CANCELLED,
      'user',
      { userId: ctx.user.id, customReason: 'Purchaser cancelled gift before delivery' }
    )
    if (!result.success) {
      throw new TRPCError({ code: 'INTERNAL_SERVER_ERROR', message: result.error })
    }
    return { success: true, message: 'Gift cancelled. A full refund has been processed.' }
  })
```

```typescript
updateGiftMessage: protectedProcedure
  .input(z.object({
    bookingId: z.string(),
    message: z.string().max(1000).optional(), // Allow empty to clear message
  }))
  .mutation(async ({ ctx, input }) => {
    // 1. Find booking and verify purchaser owns it
    const booking = await ctx.db.booking.findUnique({
      where: { id: input.bookingId },
    })
    if (!booking?.isGift || booking.giftPurchaserId !== ctx.user.id) {
      throw new TRPCError({ code: 'FORBIDDEN', message: 'Not authorized' })
    }
    // 2. Verify status is PENDING
    if (booking.giftStatus !== 'PENDING') {
      throw new TRPCError({
        code: 'BAD_REQUEST',
        message: 'Message can only be edited for pending gifts'
      })
    }
    // 3. Update message
    await ctx.db.booking.update({
      where: { id: input.bookingId },
      data: { giftMessage: input.message || null },
    })
    return { success: true, message: 'Gift message updated' }
  })
```

**resendGiftNotification:**
```typescript
resendGiftNotification: protectedProcedure
  .input(z.object({ bookingId: z.string() }))
  .mutation(async ({ ctx, input }) => {
    // 1. Check rate limit (3 per 24h per gift)
    const rateLimit = await checkRateLimit('giftResend', input.bookingId)
    if (rateLimit && !rateLimit.success) {
      throw new TRPCError({
        code: 'TOO_MANY_REQUESTS',
        message: `Too many resend attempts. Try again in ${Math.ceil((rateLimit.reset - Date.now()) / 3600000)} hours.`,
      })
    }
    // 2. Find booking and verify purchaser owns it
    const booking = await ctx.db.booking.findUnique({
      where: { id: input.bookingId },
      include: { package: true, trip: true },
    })
    if (!booking?.isGift || booking.giftPurchaserId !== ctx.user.id) {
      throw new TRPCError({ code: 'FORBIDDEN', message: 'Not authorized' })
    }
    // 3. Verify status is SENT (only SENT can resend)
    if (booking.giftStatus !== 'SENT') {
      throw new TRPCError({
        code: 'BAD_REQUEST',
        message: 'Can only resend notification for sent gifts'
      })
    }
    // 4. Send the notification email (reuse handleSentTransition logic or call sendEmail directly)
    // Get purchaser info and send email using existing template
    const purchaserInfo = await getPurchaserInfo(booking.giftPurchaserId!)
    const baseUrl = process.env.NEXT_PUBLIC_APP_URL || 'https://pickleballpassport.com'
    const acceptanceUrl = `${baseUrl}/gift/accept?token=${booking.giftAcceptanceToken}`

    const emailData: GiftNotificationRecipientData = {
      recipientFirstName: booking.giftRecipientName?.split(' ')[0] || 'Guest',
      recipientEmail: booking.giftRecipientEmail!,
      purchaserFirstName: purchaserInfo?.firstName || 'Someone special',
      purchaserLastName: purchaserInfo?.lastName || '',
      bookingReference: booking.bookingReference,
      packageName: booking.package.name,
      duration: booking.duration,
      accommodationTier: booking.accommodationTier,
      tripStartDate: booking.trip?.startDate?.toISOString(),
      tripEndDate: booking.trip?.endDate?.toISOString(),
      destination: booking.trip?.destination || 'Chiang Mai, Thailand',
      totalValue: booking.totalPrice,
      giftMessage: booking.giftMessage || undefined,
      acceptanceUrl,
    }

    const { subject, html, text } = generateGiftNotificationRecipientEmail(emailData)
    await sendEmail({
      to: booking.giftRecipientEmail!,
      subject: `[Reminder] ${subject}`,
      html,
      text,
    })

    giftLogger.info({
      msg: 'Gift notification resent',
      bookingId: booking.id,
      bookingReference: booking.bookingReference,
      recipientEmail: booking.giftRecipientEmail,
    })

    return {
      success: true,
      message: 'Notification email resent to recipient',
      remaining: rateLimit?.remaining ?? 2,
    }
  })
```
