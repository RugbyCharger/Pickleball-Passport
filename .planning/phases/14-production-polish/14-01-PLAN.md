       Example implementation in auth.tsx:
       ```typescript
       import { setOneSignalExternalId, clearOneSignalUser } from '@/lib/onesignal';
       import { useUser } from '@clerk/clerk-expo';
       import { useEffect } from 'react';

       // Add this component inside ClerkLoaded:
       function OneSignalUserSync() {
         const { user } = useUser();

         useEffect(() => {
           if (user?.id) {
             setOneSignalExternalId(user.id);
           } else {
             clearOneSignalUser();
           }
         }, [user?.id]);

         return null;
       }

       // In AuthProvider, add <OneSignalUserSync /> inside ClerkLoaded
       ```

    Pattern for initialization (from research):
    ```typescript
    import { OneSignal, LogLevel } from 'react-native-onesignal';
    import Constants from 'expo-constants';

    export function initializeOneSignal() {
      if (__DEV__) {
        OneSignal.Debug.setLogLevel(LogLevel.Verbose);
      }

      const appId = Constants.expoConfig?.extra?.oneSignalAppId;
      if (!appId) {
        console.warn('OneSignal App ID not configured');
        return;
      }

      OneSignal.initialize(appId);
    }
    ```

    Important: Do NOT request notification permission here. That's handled contextually in the app (NotificationPrompt component in Plan 02).
